# P3 — Query (search navigation + sections parser) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Given a course query (term, subject, faculty, course number), drive Minerva's advanced search, parse the "Sections Found" table into `SectionStats[]`, match the target CRN, and feed it to the P1 `decide()` engine.

**Architecture:** A pure `parseSections(html)` (cheerio) unit-tested with a synthetic fixture, plus a `QueryClient` that uses the P2 `SessionManager` to navigate term-select → advanced search → results, then parses. End-to-end verified against the live site (the session is already logged in).

**Tech Stack:** TypeScript, cheerio, Playwright (via SessionManager), Vitest.

**Privacy:** Unit tests use a SYNTHETIC fixture (no personal info). The real gitignored HTML samples are used only for local manual verification, never committed.

## Confirmed Minerva facts (from real saved pages)

- Results table: `table.datadisplaytable` with `<caption>Sections Found</caption>`.
- Subject group rows: `<th class="ddtitle">`. Header row: `<th class="ddheader">`.
- Data rows: 20 `<td class="dddefault">` cells. Column order (0-based):
  `0 Select, 1 CRN(<a>), 2 Subj, 3 Crse, 4 Sec, 5 Type, 6 Credits, 7 Title, 8 Days, 9 Time, 10 Cap, 11 Act, 12 Rem, 13 WLCap, 14 WLAct, 15 WLRem, 16 Instr, 17 Date, 18 Loc, 19 Status`.
- Seat cells are plain integers (no-waitlist renders `0`, not blank).
- NOTES rows = 2 cells (colspan 25); spacer rows = 1 cell (colspan 26) → both skipped.
- Term select: `select[name="p_term"]` (value = 6-digit code e.g. `202701`), form posts to `bwckgens.p_proc_term_date`.
- Advanced search: `select[name="sel_subj"]`, `select[name="sel_coll"]` (faculty), `input[name="sel_crse"]` (course number); form posts to `bwskfcls.P_GetCrse_Advanced`.

- [ ] **Step 3: Implement `parse-sections.ts`**

```ts
import * as cheerio from 'cheerio';
import type { SectionStats } from '@autoregister/shared';

/**
 * Parse Minerva's "Sections Found" results HTML into per-CRN seat stats.
 * Pure: no IO. Robust to NOTES/spacer/header rows.
 */
export function parseSections(html: string): SectionStats[] {
  const $ = cheerio.load(html);
  const tables = $('table.datadisplaytable').toArray();
  const table = tables.find((t) => $(t).find('caption').text().includes('Sections Found')) ?? tables[0];
  if (!table) return [];

  const out: SectionStats[] = [];
  $(table)
    .find('tr')
    .each((_, tr) => {
      const tds = $(tr).find('td');
      if (tds.length < 16) return; // header (th)/notes/spacer rows
      const crn = $(tds[1]).text().trim();
      if (!/^\d{4,5}$/.test(crn)) return;
      const n = (i: number) => Number.parseInt($(tds[i]).text().trim(), 10) || 0;
      out.push({
        crn,
        cap: n(10),
        act: n(11),
        rem: n(12),
        wlcap: n(13),
        wlact: n(14),
        wlrem: n(15),
      });
    });
  return out;
}
```

```ts
import type { CourseQuery, Decision, SectionStats } from '@autoregister/shared';
import { decide } from '@autoregister/shared';
import { MINERVA_BASE } from '../session/config';
import type { SessionManager } from '../session/session-manager';
import { parseSections } from './parse-sections';

const TERM_SELECT_URL = `${MINERVA_BASE}/bwskfcls.p_sel_crse_search`;

export interface CourseCheck {
  stats: SectionStats;
  decision: Decision;
}

/** Drives Minerva's advanced course search and parses results. */
export class QueryClient {
  constructor(private readonly session: SessionManager) {}

  /** Run the advanced search and return all parsed sections for the query. */
  async getSections(query: CourseQuery): Promise<SectionStats[]> {
    const page = await this.session.getPage();

    // 1. Select the term (form posts to p_proc_term_date).
    await page.goto(TERM_SELECT_URL, { waitUntil: 'domcontentloaded' });
    await page.selectOption('select[name="p_term"]', query.term);
    await Promise.all([
      page.waitForLoadState('domcontentloaded'),
      page.click('form[action*="p_proc_term_date"] input[type="submit"]'),
    ]);

    // 2. Fill the advanced search form (subject + course number + optional faculty).
    await page.selectOption('select[name="sel_subj"]', query.subject);
    if (query.faculty) {
      await page.selectOption('select[name="sel_coll"]', query.faculty).catch(() => undefined);
    }
    await page.fill('input[name="sel_crse"]', query.courseNumber);
    await Promise.all([
      page.waitForLoadState('domcontentloaded'),
      page.click('form[action*="P_GetCrse_Advanced"] input[type="submit"][value*="Get Course Sections"]'),
    ]);

    // 3. Parse the results.
    return parseSections(await page.content());
  }

  /** Find the target CRN's stats and run the decision engine. Null if not found. */
  async checkCourse(query: CourseQuery): Promise<CourseCheck | null> {
    const sections = await this.getSections(query);
    const stats = sections.find((s) => s.crn === query.targetCrn);
    if (!stats) return null;
    return { stats, decision: decide(stats) };
  }
}
```

```ts
import { SessionManager } from '../session/session-manager';
import { QueryClient } from './query-client';

/**
 * Manual verification:
 *   npm run query -w @autoregister/server -- <term> <subject> <courseNumber> <targetCrn> [faculty]
 * e.g. npm run query -w @autoregister/server -- 202609 COMP 551 2348
 */
async function main() {
  const [term, subject, courseNumber, targetCrn, faculty] = process.argv.slice(2);
  if (!term || !subject || !courseNumber || !targetCrn) {
    console.error('Usage: query -- <term> <subject> <courseNumber> <targetCrn> [faculty]');
    process.exit(1);
  }
  const session = new SessionManager();
  await session.launch();
  if (!(await session.isLoggedIn())) {
    console.error('Not logged in — run `npm run session:login` first.');
    process.exit(1);
  }
  const client = new QueryClient(session);
  const sections = await client.getSections({ term, subject, courseNumber, targetCrn, faculty });
  console.log('Parsed sections:', JSON.stringify(sections, null, 2));
  const check = await client.checkCourse({ term, subject, courseNumber, targetCrn, faculty });
  console.log('Target check:', JSON.stringify(check, null, 2));
  await Promise.race([session.close().catch(() => undefined), new Promise((r) => setTimeout(r, 5000))]);
  process.exit(0);
}

main().catch((e) => {
  console.error('Query script failed:', e);
  process.exit(1);
});
```
