# P4 — Register / Waitlist Actor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Given a target CRN and a decided action (REGISTER or WAITLIST), perform it on Minerva's Quick Add/Drop, detect the outcome (registered / waitlisted / closed / waitlist-full / error), handle the waitlist re-submit flow, and capture+report unknown errors.

**Architecture:** A pure `parseRegisterResult(html, crn)` (cheerio) unit-tested against the saved fixtures (no live risk), plus a `RegisterClient` that uses the P2 `SessionManager` to ensure term, submit the CRN, and re-submit for waitlist. A before/after Current-Schedule snapshot makes success detection false-positive-proof (per user).

**Tech Stack:** TypeScript, cheerio, Playwright, Vitest.

**⚠️ SAFETY:** This phase performs real registration actions. Offline parts (parser + tests) are safe. LIVE testing can actually enroll/waitlist the user — it MUST be gated on explicit user confirmation (see Task 7). The actor never runs on its own; it is invoked only by the scheduler (P5) per the per-course auto/notify mode.

```ts
import * as cheerio from 'cheerio';
import type { RegisterOutcome } from '@autoregister/shared';

function headerIdx($: cheerio.CheerioAPI, table: cheerio.Cheerio<never>): Record<string, number> {
  const idx: Record<string, number> = {};
  table
    .find('tr')
    .filter((_, tr) => $(tr).find('th.ddheader').length > 0)
    .first()
    .find('th')
    .each((i, th) => {
      const k = $(th).text().trim().toLowerCase().replace(/\s+/g, ' ');
      if (k) idx[k] = i;
    });
  return idx;
}

export function parseRegisterResult(html: string, crn: string): RegisterOutcome {
  const $ = cheerio.load(html);

  // 1. Success: CRN present in Current Schedule with a "Web Registered"/"Waitlist" status.
  const sched = $('table.datadisplaytable[summary="Current Schedule"]').first();
  if (sched.length) {
    const idx = headerIdx($, sched);
    const crnCol = idx['crn'] ?? 2;
    const statusCol = idx['status'] ?? 0;
    let found: RegisterOutcome | null = null;
    sched.find('tr').each((_, tr) => {
      const tds = $(tr).find('td');
      if (tds.length <= Math.max(crnCol, statusCol)) return;
      if ($(tds[crnCol]).text().trim() !== crn) return;
      const status = $(tds[statusCol]).text().trim();
      if (/waitlist/i.test(status)) found = { kind: 'waitlisted', crn, message: status };
      else if (/web registered|registered/i.test(status)) found = { kind: 'registered', crn, message: status };
    });
    if (found) return found;
  }

  // 2. Errors: CRN present in the Registration Add Errors table.
  const errs = $('table.datadisplaytable[summary*="Registration Errors"]').first();
  if (errs.length) {
    const idx = headerIdx($, errs);
    const crnCol = idx['crn'] ?? 2;
    const statusCol = idx['status'] ?? 0;
    const actionCol = idx['action'] ?? 1;
    let found: RegisterOutcome | null = null;
    errs.find('tr').each((_, tr) => {
      const tds = $(tr).find('td');
      if (tds.length <= Math.max(crnCol, statusCol, actionCol)) return;
      if ($(tds[crnCol]).text().trim() !== crn) return;
      const message = $(tds[statusCol]).text().trim();
      const canWaitlist = $(tds[actionCol]).find('option[value="LW"]').length > 0;
      if (canWaitlist || /reserved for waitlist/i.test(message)) found = { kind: 'waitlist-available', crn, message };
      else if (/waitlist full/i.test(message)) found = { kind: 'waitlist-full', crn, message };
      else if (/closed|class full|full/i.test(message)) found = { kind: 'closed', crn, message };
      else found = { kind: 'error', crn, message };
    });
    if (found) return found;
  }

  return { kind: 'not-found', crn };
}
```

```ts
import type { ActionKind, RegisterOutcome } from '@autoregister/shared';
import type { SessionManager } from '../session/session-manager';
import { ADD_DROP_TERM_URL, QUICK_ADD_URL } from '../session/config';
import { parseRegisterResult } from './parse-register-result';

export class RegisterClient {
  constructor(private readonly session: SessionManager) {}

  /** Ensure the Quick Add/Drop term matches `term` (term_in -> P_StoreTerm). */
  private async ensureTerm(term: string): Promise<void> {
    const page = await this.session.getPage();
    await page.goto(ADD_DROP_TERM_URL, { waitUntil: 'domcontentloaded' });
    // Page may go straight to the worksheet if a term is already set; only set if the select exists.
    if (await page.locator('select[name="term_in"]').count()) {
      await page.selectOption('select[name="term_in"]', term);
      await Promise.all([
        page.waitForLoadState('domcontentloaded'),
        page.click('form[action*="P_StoreTerm"] input[type="submit"]'),
      ]);
    }
  }

  /**
   * Perform `action` for `crn` in `term`. Handles the waitlist re-submit flow.
   * Returns the final outcome. Never throws on a normal Minerva error.
   */
  async act(term: string, crn: string, action: ActionKind): Promise<RegisterOutcome> {
    if (action === 'NOOP') return { kind: 'not-found', crn };
    await this.ensureTerm(term);
    const page = await this.session.getPage();
    await page.goto(QUICK_ADD_URL, { waitUntil: 'domcontentloaded' });

    // Enter the CRN in the first empty worksheet field and submit.
    const crnInput = page.locator('input[name="CRN_IN"]').last();
    await crnInput.fill(crn);
    await Promise.all([
      page.waitForLoadState('domcontentloaded'),
      page.click('input[name="REG_BTN"][value="Submit Changes"]'),
    ]);

    let outcome = parseRegisterResult(await page.content(), crn);

    // Waitlist flow: select "Add to Waitlist" (LW) and submit again.
    if (action === 'WAITLIST' && outcome.kind === 'waitlist-available') {
      const ok = await page
        .selectOption(`select[name="RSTS_IN"] >> nth=-1`, 'LW')
        .then(() => true)
        .catch(() => false);
      if (ok) {
        await Promise.all([
          page.waitForLoadState('domcontentloaded'),
          page.click('input[name="REG_BTN"][value="Submit Changes"]'),
        ]);
        outcome = parseRegisterResult(await page.content(), crn);
      }
    }
    return outcome;
  }
}
```
